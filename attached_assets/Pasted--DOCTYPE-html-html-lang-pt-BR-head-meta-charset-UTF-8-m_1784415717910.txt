    <script>
        // Variáveis globais
        let currentUnit = 'swat';
        let currentUnitName = 'SWAT';
        let currentAccentColor = '#d4af37';
        
        // Função para gerar código de operação aleatório
        function generateOperationCode() {
            const letters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
            const numbers = '0123456789';
            let code = '';
            
            // 2 letras aleatórias
            for (let i = 0; i < 2; i++) {
                code += letters.charAt(Math.floor(Math.random() * letters.length));
            }
            
            // 4 números aleatórios
            for (let i = 0; i < 4; i++) {
                code += numbers.charAt(Math.floor(Math.random() * numbers.length));
            }
            
            // 2 letras aleatórias
            for (let i = 0; i < 2; i++) {
                code += letters.charAt(Math.floor(Math.random() * letters.length));
            }
            
            return code;
        }

        // Função para gerar o relatório com os dados inseridos
        function generateReport() {
            // Obter valores dos campos
            const agentName = document.getElementById('agentName').value || '[NÃO INFORMADO]';
            const operationDate = formatDateTime(document.getElementById('operationDate').value) || '[NÃO INFORMADO]';
            let operationCode = document.getElementById('operationCode').value;
            if (!operationCode) {
                operationCode = generateOperationCode();
                document.getElementById('operationCode').value = operationCode;
            }
            const reportTitle = document.getElementById('reportTitle').value || '[NÃO INFORMADO]';
            const reportDetails = document.getElementById('reportDetails').value || '[NÃO INFORMADO]';
            const agentSignature = document.getElementById('agentSignature').value || 'Assinatura do Operador';
            const confidentialLevel = document.getElementById('confidentialLevel').value;
            
            // Atualizar o relatório
            document.getElementById('displayAgentName').textContent = agentName;
            document.getElementById('displayOperationDate').textContent = operationDate;
            document.getElementById('displayOperationCode').textContent = operationCode;
            document.getElementById('displayReportTitleHeader').textContent = reportTitle;
            document.getElementById('displayReportDetails').textContent = reportDetails;
            document.getElementById('displaySignature').textContent = agentSignature;
            
            // Gerar número de relatório aleatório
            const randomNum = Math.floor(Math.random() * 1000);
            const today = new Date();
            document.getElementById('reportNumber').textContent = `${currentUnit.toUpperCase()}-${today.getFullYear()}-${randomNum.toString().padStart(3, '0')}`;
            
            // Atualizar selo de confidencialidade
            const stamp = document.getElementById('confidentialStamp');
            stamp.textContent = '';
            stamp.style.color = currentAccentColor;
            
            if (confidentialLevel === 'confidential') {
                stamp.textContent = 'CONFIDENCIAL';
                stamp.style.opacity = '0.2';
            } else if (confidentialLevel === 'topsecret') {
                stamp.textContent = 'ULTRA-SECRETO';
                stamp.style.opacity = '0.3';
            }
            
            // Atualizar watermark
            document.getElementById('watermark').textContent = currentUnitName;
        }
        
        // Função para exportar como imagem
        function exportAsImage() {
            generateReport(); // Atualiza o relatório antes de exportar
            
            // Criar um clone do elemento para captura
            const element = document.getElementById('tacticalReport');
            const clone = element.cloneNode(true);
            clone.style.position = 'absolute';
            clone.style.left = '-9999px';
            clone.style.top = '0';
            clone.style.zIndex = '9999';
            document.body.appendChild(clone);
            
            // Forçar estilos no clone para garantir que apareçam corretamente na imagem
            clone.style.width = element.offsetWidth + 'px';
            clone.style.height = 'auto';
            clone.style.background = window.getComputedStyle(element).background;
            
            // Garantir que o texto está visível no clone
            const textElements = clone.querySelectorAll('*');
            textElements.forEach(el => {
                const computedStyle = window.getComputedStyle(el);
                el.style.color = computedStyle.color;
                el.style.backgroundColor = computedStyle.backgroundColor;
                el.style.fontFamily = computedStyle.fontFamily;
                el.style.fontSize = computedStyle.fontSize;
                el.style.fontWeight = computedStyle.fontWeight;
            });
            
            // Configurações do html2canvas para melhor qualidade
            html2canvas(clone, {
                scale: 2,
                logging: false,
                useCORS: true,
                allowTaint: true,
                backgroundColor: null,
                letterRendering: true,
                onclone: function(clonedDoc) {
                    // Garantir que os estilos são aplicados no clone
                    const clonedElement = clonedDoc.getElementById('tacticalReport');
                    clonedElement.style.width = element.offsetWidth + 'px';
                    clonedElement.style.height = 'auto';
                }
            }).then(canvas => {
                const link = document.createElement('a');
                link.download = `relatorio-${currentUnit}.png`;
                link.href = canvas.toDataURL('image/png');
                link.click();
                
                // Remover o clone após a captura
                document.body.removeChild(clone);
            });
        }
        
        // Função para formatar data/hora
        function formatDateTime(datetimeStr) {
            if (!datetimeStr) return '';
            
            const date = new Date(datetimeStr);
            return date.toLocaleString('pt-BR', {
                day: '2-digit',
                month: '2-digit',
                year: 'numeric',
                hour: '2-digit',
                minute: '2-digit'
            });
        }
        
        // Função para mudar o tema
        function changeTheme(mainColor, secondaryColor) {
            document.documentElement.style.setProperty('--main-color', mainColor);
            document.documentElement.style.setProperty('--secondary-color', secondaryColor);
            
            // Atualizar ícones
            const icons = document.querySelectorAll('.field-icon');
            icons.forEach(icon => {
                icon.style.backgroundColor = mainColor;
            });
            
            // Atualizar seleção de tema
            document.querySelectorAll('.theme-option').forEach(option => {
                option.classList.remove('selected');
            });
            event.target.classList.add('selected');
            
            generateReport();
        }
        
        // Função para mudar a unidade
        function changeUnit(unit, unitName, mainColor, secondaryColor, accentColor) {
            currentUnit = unit;
            currentUnitName = unitName;
            currentAccentColor = accentColor;
            
            // Atualizar cores
            document.documentElement.style.setProperty('--main-color', mainColor);
            document.documentElement.style.setProperty('--secondary-color', secondaryColor);
            document.documentElement.style.setProperty('--accent-color', accentColor);
            
            // Atualizar interface
            document.getElementById('unitBadge').textContent = unitName;
            document.getElementById('unitTitle').textContent = `UNIDADE TÁTICA ESPECIAL - ${unitName}`;
            document.getElementById('unitFooter').textContent = `${unitName} - DEPARTAMENTO DE DEFESA`;
            document.getElementById('watermark').textContent = unitName;
            
            // Atualizar ícones
            const icons = document.querySelectorAll('.field-icon');
            icons.forEach(icon => {
                icon.style.backgroundColor = mainColor;
            });
            
            // Ativar botão selecionado
            document.querySelectorAll('.unit-option').forEach(opt => opt.classList.remove('active'));
            event.target.classList.add('active');
            
            generateReport();
        }
        
        // Gerar relatório inicial
        window.onload = function() {
            // Definir data/hora atual como padrão
            const now = new Date();
            const year = now.getFullYear();
            const month = String(now.getMonth() + 1).padStart(2, '0');
            const day = String(now.getDate()).padStart(2, '0');
            const hours = String(now.getHours()).padStart(2, '0');
            const minutes = String(now.getMinutes()).padStart(2, '0');
