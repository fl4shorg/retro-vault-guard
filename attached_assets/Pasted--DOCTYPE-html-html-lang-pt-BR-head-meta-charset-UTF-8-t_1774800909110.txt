<script>
const API_BASE = "https://www.api.neext.online/api/chat";

let userId = localStorage.getItem("uid") || Date.now().toString();
localStorage.setItem("uid", userId);

let user = localStorage.getItem("nome") || "";
let replyData = null;

// LOGIN
function entrar() {
  const nome = document.getElementById("nomeInput").value.trim();
  if(!nome) return alert("Digite um nome");
  user = nome;
  localStorage.setItem("nome", user);
  document.getElementById("login").style.display="none";
}

if(user) document.getElementById("login").style.display="none";

// REPLY
function responderMsg(id, nome, texto) {
  replyData = { id, nome, texto };
  renderReplyPreview();
}

function renderReplyPreview() {
  const div = document.getElementById("replyPreview");
  if(!replyData){
    div.innerHTML="";
    return;
  }
  div.innerHTML = `
    <div class="reply-preview">
      <div>
        <b>${replyData.nome}</b><br>
        ${replyData.texto}
      </div>
      <button onclick="cancelarReply()">X</button>
    </div>
  `;
}

function cancelarReply() {
  replyData = null;
  renderReplyPreview();
}

// LISTAR MENSAGENS
async function carregarMensagens() {
  try {
    const res = await fetch(API_BASE);
    const msgs = await res.json();
    const container = document.getElementById("msgs");
    container.innerHTML = "";

    msgs.forEach(d => {
      let html = `<div class="msg ${d.userId===userId?'me':''} ${d.apagado?'deleted':''}">`;

      if(d.reply){
        html += `<div class="reply-box"><b>${d.reply.nome}</b><br>${d.reply.texto}</div>`;
      }

      html += `<b>${d.user}</b><br>`;
      html += d.apagado ? "🚫 Mensagem apagada" : `${d.texto} ${d.editado ? '<small>(editada)</small>' : ''}`;

      html += `<div class="actions">`;
      if(!d.apagado) html += `<button class="reply" onclick="responderMsg('${d.id}','${d.user}','${(d.texto||"").replace(/'/g,"\\'")}')">Responder</button>`;
      if(d.userId===userId && !d.apagado){
        html += `<button class="edit" onclick="editarMsg('${d.id}','${d.texto.replace(/'/g,"\\'")}')">Editar</button>`;
        html += `<button class="delete" onclick="deletarMsg('${d.id}')">Excluir</button>`;
      }
      html += `</div></div>`;
      container.innerHTML += html;
    });

    container.scrollTop = container.scrollHeight;
  } catch(err){
    console.error(err);
  }
}

// ENVIAR MENSAGEM
async function enviarTexto() {
  const input = document.getElementById("texto");
  const texto = input.value.trim();
  if(!texto) return;

  await fetch(API_BASE, {
    method:"POST",
    headers:{"Content-Type":"application/json"},
    body: JSON.stringify({ user, userId, texto, reply: replyData })
  });

  input.value="";
  replyData = null;
  renderReplyPreview();
  carregarMensagens();
}

// EDITAR
async function editarMsg(id, textoAntigo){
  const novo = prompt("Editar mensagem:", textoAntigo);
  if(!novo || novo.trim()==="") return;
  await fetch(`${API_BASE}/${id}`,{
    method:"PUT",
    headers:{"Content-Type":"application/json"},
    body: JSON.stringify({ texto: novo })
  });
  carregarMensagens();
}

// DELETE (soft)
async function deletarMsg(id){
  if(!confirm("Excluir mensagem?")) return;
  await fetch(`${API_BASE}/${id}`, { method:"DELETE" });
  carregarMensagens();
}

// atualizar a cada 2s
setInterval(carregarMensagens, 2000);
carregarMensagens();
</script>
